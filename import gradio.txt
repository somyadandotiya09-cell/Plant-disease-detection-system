
import gradio as gr
import tensorflow as tf
import numpy as np
import json
from PIL import Image
import os

# ---------------------------------
# 1. Load trained disease model
# ---------------------------------

MODEL_PATH = "plant_disease_model.keras"
CLASS_PATH = "class_names.json"

model = tf.keras.models.load_model(MODEL_PATH)

# ---------------------------------
# 2. Load disease class names
# ---------------------------------

with open(CLASS_PATH, "r") as file:
    class_names = json.load(file)

if isinstance(class_names, dict):
    class_names = list(class_names.values())

# ---------------------------------
# 3. Image preprocessing
# ---------------------------------

def preprocess_image(image):

    image = image.convert("RGB")

    image = image.resize((128, 128))

    image_array = np.array(image)

    image_array = image_array.astype("float32") / 255.0

    image_array = np.expand_dims(image_array, axis=0)

    return image_array

# ---------------------------------
# 4. Disease prediction
# ---------------------------------

def predict_disease(image):

    if image is None:
        return "Please upload a leaf image.", ""

    try:
        processed_image = preprocess_image(image)

        predictions = model.predict(
            processed_image,
            verbose=0
        )

        predicted_index = int(np.argmax(predictions[0]))

        confidence = float(
            predictions[0][predicted_index]
        ) * 100

        disease_name = str(class_names[predicted_index])

        if "healthy" in disease_name.lower():
            result = "Healthy Plant"
            advice = "The plant appears healthy. Continue regular care."
        else:
            result = disease_name
            advice = (
                "Disease detected. Consider isolating the affected "
                "plant and consulting an agricultural expert."
            )

        output = f"""
        DISEASE DETECTION RESULT

        Disease: {result}

        Confidence: {confidence:.2f}%

        Recommendation: {advice}
        """

        return output, result

    except Exception as error:

        return f"Detection Error: {str(error)}", "Error"

# ---------------------------------
# 5. Website interface
# ---------------------------------

with gr.Blocks(
    theme=gr.themes.Soft(),
    title="Plant Disease Detection System"
) as demo:

    gr.Markdown(
        """
        # 🌿 Plant Disease Detection System

        ### AI-Based Plant Disease Identification

        Upload a picture of a plant leaf to identify
        possible diseases using Deep Learning.
        """
    )

    with gr.Row():

        with gr.Column():

            image_input = gr.Image(
                type="pil",
                label="Upload Plant Leaf Image"
            )

            detect_button = gr.Button(
                "Detect Disease",
                variant="primary"
            )

            clear_button = gr.ClearButton()

        with gr.Column():

            result_output = gr.Textbox(
                label="Detection Result",
                lines=7
            )

            disease_output = gr.Textbox(
                label="Predicted Disease"
            )

    detect_button.click(
        fn=predict_disease,
        inputs=image_input,
        outputs=[result_output, disease_output]
    )

    clear_button.click(
        fn=lambda: (None, "", ""),
        inputs=[],
        outputs=[
            image_input,
            result_output,
            disease_output
        ]
    )

    gr.Markdown(
        """
        ---
        **Project:** Plant Disease Detection System

        **Technology:** Python, TensorFlow, CNN and Gradio

        **Purpose:** Early identification of plant diseases
        through leaf image classification.
        """
    )

# ---------------------------------
# 6. Launch website
# ---------------------------------

if __name__ == "__main__":

    demo.launch(share=True)
